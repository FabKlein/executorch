cmake_minimum_required(VERSION 3.19)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

# Source root directory for executorch
if(NOT EXECUTORCH_ROOT)
  set(EXECUTORCH_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
endif()

include(${EXECUTORCH_ROOT}/tools/cmake/Utils.cmake)
include(${EXECUTORCH_ROOT}/tools/cmake/Codegen.cmake)
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

# CMSIS-NN configuration with dynamic path detection
set(CMSIS_NN_VERSION
    "dbf45dbfcc515421dd6099037d3e2637b90748c8"
    CACHE STRING "CMSIS-NN version to download"
)
set(CMSIS_NN_LOCAL_PATH
    ""
    CACHE PATH "Path to existing local CMSIS-NN installation"
)
option(CORTEX_M_ENABLE_RUNTIME_CHECKS
       "Enable additional Cortex-M runtime assertions and validation checks"
       OFF
)
option(EXECUTORCH_CORTEX_M_CMSIS_NN_DEBUG_SYMBOLS
       "Build CMSIS-NN with debug symbols for FVP source coverage/profiling"
       OFF
)

set(CMSISNN_BUILD_PYBIND
    ${EXECUTORCH_BUILD_CMSIS_NN_PYBINDS}
    CACHE BOOL "Build CMSIS-NN Python bindings" FORCE
)

# Try to find existing / local CMSIS-NN installation. This is useful for
# debugging and testing with local changes. This is not common, as the CMSIS-NN
# library is downloaded via FetchContent in the default/regular case.
if(CMSIS_NN_LOCAL_PATH AND EXISTS "${CMSIS_NN_LOCAL_PATH}")
  message(STATUS "Using CMSIS-NN from specified path: ${CMSIS_NN_LOCAL_PATH}")
  add_subdirectory(
    ${CMSIS_NN_LOCAL_PATH} ${CMAKE_CURRENT_BINARY_DIR}/cmsis_nn-build
  )
else()
  # Use FetchContent with automatic fallback
  message(STATUS "Using CMSIS-NN via FetchContent")

  FetchContent_Declare(
    cmsis_nn
    GIT_REPOSITORY https://github.com/ARM-software/CMSIS-NN.git
    GIT_TAG ${CMSIS_NN_VERSION}
    SOURCE_DIR ${CMAKE_CURRENT_BINARY_DIR}/cmsis_nn-src BINARY_DIR
    ${CMAKE_CURRENT_BINARY_DIR}/cmsis_nn-build SUBBUILD_DIR
    ${CMAKE_CURRENT_BINARY_DIR}/cmsis_nn-subbuild
  )

  FetchContent_MakeAvailable(cmsis_nn)
endif()

if(TARGET cmsis-nn)
  if(CMSIS_NN_LOCAL_PATH AND EXISTS "${CMSIS_NN_LOCAL_PATH}")
    set(cmsis_nn_source_dir "${CMSIS_NN_LOCAL_PATH}")
  else()
    set(cmsis_nn_source_dir "${cmsis_nn_SOURCE_DIR}")
  endif()
  if(cmsis_nn_source_dir)
    set(cmsis_nn_include_dir "${cmsis_nn_source_dir}/Include")
    set_target_properties(
      cmsis-nn
      PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES
        "$<BUILD_INTERFACE:${cmsis_nn_include_dir}>;$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/cmsis-nn>"
    )
    install(DIRECTORY "${cmsis_nn_include_dir}/"
            DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/cmsis-nn"
    )
  endif()
endif()

if(EXECUTORCH_BUILD_CMSIS_NN_PYBINDS)
  if(NOT TARGET cmsis_nn)
    message(
      FATAL_ERROR
        "EXECUTORCH_BUILD_CMSIS_NN_PYBINDS is ON, but CMSIS-NN did not define the cmsis_nn pybind target."
    )
  endif()
endif()

if(EXECUTORCH_CORTEX_M_CMSIS_NN_DEBUG_SYMBOLS AND TARGET cmsis-nn)
  message(STATUS "Building CMSIS-NN with debug symbols enabled")
  target_compile_options(cmsis-nn PRIVATE -g)
endif()

# These backend switches mirror the CMSIS-NN build and drive both source
# selection and the capability artifact consumed by Python lowering.
set(EXECUTORCH_CORTEX_M_ENABLE_FLOAT32
    ${ARM_NN_ENABLE_F32}
    CACHE BOOL "Enable Cortex-M CMSIS-NN float32 kernels and lowering" FORCE
)
set(EXECUTORCH_CORTEX_M_ENABLE_FLOAT16
    ${ARM_NN_ENABLE_F16}
    CACHE BOOL "Enable Cortex-M CMSIS-NN float16 kernels and lowering" FORCE
)

if(EXECUTORCH_CORTEX_M_ENABLE_FLOAT32)
  set(_cortex_m_enable_f32_json "true")
  set(_cortex_m_enable_f32_arg "ON")
else()
  set(_cortex_m_enable_f32_json "false")
  set(_cortex_m_enable_f32_arg "OFF")
endif()
if(EXECUTORCH_CORTEX_M_ENABLE_FLOAT16)
  set(_cortex_m_enable_f16_json "true")
  set(_cortex_m_enable_f16_arg "ON")
else()
  set(_cortex_m_enable_f16_json "false")
  set(_cortex_m_enable_f16_arg "OFF")
endif()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/float_capabilities.json"
     "{\n"
     "  \"_comment\": \"Auto-generated by CMake. Do not edit manually.\",\n"
     "  \"enable_f32\": ${_cortex_m_enable_f32_json},\n"
     "  \"enable_f16\": ${_cortex_m_enable_f16_json}\n"
     "}\n"
)

if(EXECUTORCH_BUILD_CORTEX_M)
  # Cortex-M ops kernel sources
  set(_cortex_m_kernels__srcs
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_dequantize_per_tensor.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_maximum.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_minimum.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_pad.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantize_per_tensor.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_activation.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_add.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_avg_pool2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_batch_matmul.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_conv2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_depthwise_conv2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_div.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_linear.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_max_pool2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_mul.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_quantized_transpose_conv2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_softmax.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_transpose.cpp
  )

  if(EXECUTORCH_CORTEX_M_ENABLE_FLOAT32 OR EXECUTORCH_CORTEX_M_ENABLE_FLOAT16)
    list(
      APPEND
      _cortex_m_kernels__srcs
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_activation.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_add.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_avg_pool2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_batch_matmul.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_batch_norm.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_conv2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_linear.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_lstm.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_max_pool2d.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_maximum.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_minimum.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_mul.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_pad.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_softmax.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_svdf.cpp
      ${CMAKE_CURRENT_SOURCE_DIR}/ops/op_float_transpose_conv2d.cpp
    )
  endif()

  # Generate registration only for the floating-point variants enabled in the
  # same build, preventing Python/C++ capability drift.
  set(_yaml_file ${CMAKE_CURRENT_BINARY_DIR}/ops/operators.filtered.yaml)
  file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/ops)
  execute_process(
    COMMAND
      ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_LIST_DIR}/ops/filter_float_operators_yaml.py --input
      ${CMAKE_CURRENT_LIST_DIR}/ops/operators.yaml --output ${_yaml_file}
      --enable-f32 ${_cortex_m_enable_f32_arg} --enable-f16
      ${_cortex_m_enable_f16_arg}
    COMMAND_ERROR_IS_FATAL ANY
  )

  gen_selected_ops(LIB_NAME "cortex_m_ops_lib" OPS_SCHEMA_YAML "${_yaml_file}")
  generate_bindings_for_kernels(
    LIB_NAME "cortex_m_ops_lib" CUSTOM_OPS_YAML "${_yaml_file}"
  )

  # Build library for cortex_m_kernels
  add_library(cortex_m_kernels ${_cortex_m_kernels__srcs})

  target_link_libraries(
    cortex_m_kernels
    PRIVATE cmsis-nn
    PRIVATE executorch_core
    PRIVATE kernels_util_all_deps
  )
  target_compile_definitions(
    cortex_m_kernels
    PRIVATE
      $<$<BOOL:${CORTEX_M_ENABLE_RUNTIME_CHECKS}>:CORTEX_M_ENABLE_RUNTIME_CHECKS>
      ARM_NN_ENABLE_F32=$<BOOL:${EXECUTORCH_CORTEX_M_ENABLE_FLOAT32}>
      ARM_NN_ENABLE_F16=$<BOOL:${EXECUTORCH_CORTEX_M_ENABLE_FLOAT16}>
      EXECUTORCH_CORTEX_M_ENABLE_FLOAT32=$<BOOL:${EXECUTORCH_CORTEX_M_ENABLE_FLOAT32}>
      EXECUTORCH_CORTEX_M_ENABLE_FLOAT16=$<BOOL:${EXECUTORCH_CORTEX_M_ENABLE_FLOAT16}>
  )

  target_include_directories(
    cortex_m_kernels PRIVATE ${EXECUTORCH_ROOT}/..
                             ${EXECUTORCH_ROOT}/runtime/core/portable_type/c10
  )

  gen_operators_lib(
    LIB_NAME "cortex_m_ops_lib" KERNEL_LIBS cortex_m_kernels DEPS
    executorch_core
  )

  if(CMAKE_CXX_COMPILER_ID STREQUAL "ARMClang" OR CMAKE_CXX_COMPILER_ID STREQUAL
                                                   "Clang"
  )
    target_compile_definitions(
      cortex_m_ops_lib
      PRIVATE ET_REGISTER_KERNELS_FN=executorch_arm_register_cortex_m_ops
    )
  endif()

  install(
    TARGETS cortex_m_kernels cortex_m_ops_lib cmsis-nn
    EXPORT ExecuTorchTargets
    DESTINATION ${CMAKE_INSTALL_LIBDIR}
    PUBLIC_HEADER
      DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/executorch/backends/cortex_m/ops/
  )
endif()
